class Solution {
    public List<List<Integer>> threeSum(int[] nums) {
       List<List<Integer> ans =new ArrayList<>();
       int n=nums.length;
       for(int i=0;i<n-2;i++)
       {
        int l=i+1,r=n-1;
        while(l<r)
        {
            int sum=nums[i]+nums[l]+nums[r];
            if(sum==0)
            {
                List<Integer> row=new ArrayList<>();
                row.add(nums[i]);
                row.add(nums[l]);
                row.add(nums[r]);
                ans.add(row);
            }
        }
       } 
       return ans;
    }
}
