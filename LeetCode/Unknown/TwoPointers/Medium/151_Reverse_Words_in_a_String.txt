class Solution {
    public String reverseWords(String s) {
        StringBuilder sb=new StringBuilder();
        for(int i=s.length();i>=0;i--)
        {
            char ch= s.charAt(i);
            sb.append(ch);
        }
        return sb.toString();
    }
}
