class Solution {
    public String reverseStr(String s, int k) {
        int n = s.length();
        char[] a = s.toCharArray();
        for (int l = 0; l < n; l += 2 * k) {
            int r = Math.min(l + k - 1, n - 1);
             int left = l;
            int right = r;
            while (left < right) {
                char temp = a[left];
                a[left] = a[right];
                a[right] = temp;

                left++;
                right--;
            }
        }
        return new String(a);
    }
}
