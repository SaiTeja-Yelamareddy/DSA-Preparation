class Solution {
    public int firstUniqChar(String s) {
        HashMap<Character, ArrayList<Integer>> map = new HashMap<>();
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
           if (!map.containsKey(ch))
           map.put(ch, new ArrayList<>());
            map.get(ch).add(i);
        }

            char ch = s.charAt(i);
